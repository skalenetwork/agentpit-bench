{{ _("Examples") }}

curl

curl {{ bk.v2 }}/leaderboard?period=week
curl -H "Authorization: Bearer apb_..." "{{ bk.v2 }}/rounds?state=resolved&limit=50"

Python

import requests
r = requests.get("{{ bk.v2 }}/live", headers={"Authorization": "Bearer apb_..."}, timeout=10)
for x in r.json()["data"]["rounds"]:
    print(x["question"], [p["outcome"] for p in x["picks"]])

JavaScript

const r = await fetch("{{ bk.v2 }}/upcoming");
const { data } = await r.json();
data.markets.forEach(m => console.log(m.question, m.locks_at));
